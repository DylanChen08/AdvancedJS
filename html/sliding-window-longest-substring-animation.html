测试用例

  • 1. right 从 0 到 n - 1，只扫描一遍字符串。
  • 2. Map 保存每个字符最后出现的位置。
  • 3. 如果 last[ch] >= left，说明重复字符在当前窗口内，left = last[ch] + 1。
  • 4. 每轮更新窗口长度和最大值，整体时间复杂度 O(n)。
var lengthOfLongestSubstring = function(s) {
  let left = 0;
  let ans = 0;
  const last = new Map();

  for (let right = 0; right < s.length; right++) {
    const ch = s[right];

    if (last.has(ch) && last.get(ch) >= left) {
      left = last.get(ch) + 1; // 直接跳过重复字符
    }

    last.set(ch, right);
    ans = Math.max(ans, right - left + 1);
  }

  return ans;
};
